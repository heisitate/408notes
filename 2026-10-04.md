## 算法总结
1. 关于建立一个树（采用父子表示法构造，并求每层的节点权值之和）  ${BFS}{}$
```C++
#include<cstdio>

#include<vector>

#include<queue>

using namespace std;

  

const int MAXN=1000;

  

struct Node{

    int data;

    vector<int> children;

}nodes[MAXN];

  

int main(){

    int n,k,child;

    scanf("%d",&n);

    for(int i=0;i<n;i++){

        scanf("%d",&nodes[i].data);

    }

    for(int i=0;i<n;i++){

        scanf("%d",&k);

            for(int j=0;j<k;j++){

                scanf("%d",&child);

                nodes[i].children.push_back(child);

            }

    }

    queue<int> q;

    q.push(0);

    while(!q.empty()){

        int levelsize=q.size();

        int levelsum=0;

  

        for(int i=0;i<levelsize;i++){

            int u=q.front();

            q.pop();

            levelsum+=nodes[u].data;

            for(int j=0;j<(int)nodes[u].children.size();j++){

                q.push(nodes[u].children[j]);

            }

        }

        printf("%d\n",levelsum);

    }

    return 0;

}
```
 ${DFS}{}$
 ```C++
#include<cstdio>

#include<vector>

using namespace std;

  

const int MAXN=1000;

  

struct Node{

    int data;

    vector<int> children;

}nodes[MAXN];

  

int levelsum[MAXN];

int maxlevel=0;

  

void dfs(int u,int level){

    levelsum[level]+=nodes[u].data;

    if(level>maxlevel){

        maxlevel=level;

    }

    for(int i=0;i<(int)nodes[u].children.size();i++){

        dfs(nodes[u].children[i],level+1);

    }

}

  

int main(){

    int n,k,child;

    scanf("%d",&n);

    for(int i=0;i<n;i++){

        scanf("%d",&nodes[i].data);

    }

    for(int i=0;i<n;i++){

        scanf("%d",&k);

        for(int j=0;j<k;j++){

            scanf("%d",&child);

            nodes[i].children.push_back(child);

        }

    }

    for(int i=0;i<n;i++){

        levelsum[i]=0;

    }

    dfs(0,0);

    for(int i=0;i<=maxlevel;i++){

        printf("%d\n",levelsum[i]);

    }

    return 0;

}
 ```
